#include <stdio.h>
void gcd(int a,int b)
{
   int i,gcd;
   for(i=1;i<=a && i<=b; i++)
   {
       if(a%i==0 && b%i==0)
       gcd=i;
   }
   printf("the gcd of x and y is %d",gcd);
}
int main()
{
   int x,y;
   printf("enter value of x:");
   scanf("%d",&x);
   printf("enter value of y:");
   scanf("%d",&y);
   gcd(x,y);
}
