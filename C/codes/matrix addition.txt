#include <stdio.h>

int main()
{
    int m1[3][3]= {{2,4,6},{8,10,12},{14,16,18}},m2[3][3]= {{1,2,3},{4,5,6},{7,8,9}};
	int i,j,m3[3][3];
	printf("the fist matrix is:\n");
	for (i = 0; i < 3; i++) 
	{
		for (j = 0; j < 3; j++) 
		{
			printf("%d ", m1[i][j]);
		}
		printf("\n");
	}
	printf("the second matrix is:\n");
	for (i = 0; i < 3; i++) 
	{
		for (j = 0; j < 3; j++) 
		{
			printf("%d ", m2[i][j]);
		}
		printf("\n");
	}
	for (i = 0; i < 3; i++) 
	{
		for (j = 0; j < 3; j++) 
		{
			m3[i][j] = m1[i][j] + m2[i][j];
		}
	}
	printf("The addition of the two matrices is: \n");
	for (i = 0; i < 3; i++) 
	{
		for (j = 0; j < 3; j++) 
		{
			printf("%d ", m3[i][j]);
		}
		printf("\n");
	}
}
