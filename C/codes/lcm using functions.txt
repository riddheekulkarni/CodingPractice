#include <stdio.h>
void lcm(int a,int b)
{
   int i,gcd,lcm;
   for(i=1;i<=a && i<=b; i++)
   {
       if(a%i==0 && b%i==0)
       gcd=i;
   }
   lcm=(a*b)/gcd;
   printf("lcm of x and y is %d",lcm);
}
int main()
{
   int x,y;
   printf("enter value of x:");
   scanf("%d",&x);
   printf("enter value of y:");
   scanf("%d",&y);
   lcm(x,y);
}
